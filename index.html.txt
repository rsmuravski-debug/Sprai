<script>
function calculateSettings() {
    const inputField = document.getElementById('liters');
    const outputDiv = document.getElementById('output');
    outputDiv.innerHTML = '';

    const normaTarget = parseFloat(inputField.value);
    if (!normaTarget || normaTarget <= 0) {
        outputDiv.innerHTML = '<div class="error-message">❌ Будь ласка, введіть коректну кількість літрів!</div>';
        return;
    }

    const vidstan = 10.0 / 20; // 0.5 метра
    const idealSpeed = 9.0;

    // Ваші точні дані з фото
    const myFixedData = {
        "ЖОВТІ": [
            {"p": 2.0, "q": 0.625},
            {"p": 2.5, "q": 0.775},
            {"p": 3.0, "q": 0.875}
        ],
        "СИНІ": [
            {"p": 2.0, "q": 1.125},
            {"p": 2.5, "q": 1.250},
            {"p": 3.0, "q": 1.350}
        ]
    };

    let vsiVarianty = [];

    // Обчислення варіантів
    for (let color in myFixedData) {
        myFixedData[color].forEach(m => {
            let speedNeeded = (m.q * 600) / (vidstan * normaTarget);
            let deviation = Math.abs(speedNeeded - idealSpeed);
            
            vsiVarianty.push({
                color: color,
                pressure: m.p,
                flow: m.q,
                speed: speedNeeded,
                dev: deviation
            });
        });
    }

    // Сортування (ближче до 9 км/год — вище)
    vsiVarianty.sort((a, b) => a.dev - b.dev);

    // Відбір топ-2 варіантів у межах розумних швидкостей
    let recommended = [];
    vsiVarianty.forEach(v => {
        if (v.speed >= 5.0 && v.speed <= 13.5 && recommended.length < 2) {
            recommended.push(v);
        }
    });

    if (recommended.length === 0) {
        outputDiv.innerHTML = '<div class="error-message">⚠️ На ваших режимах форсунок неможливо підібрати адекватну швидкість трактора для такої норми виливу!</div>';
        return;
    }

    // Виведення результатів на екран телефону
    let htmlContent = '';
    recommended.forEach((v, index) => {
        let cardClass = index === 0 ? 'card best' : 'card second';
        let badgeClass = index === 0 ? 'badge badge-best' : 'badge badge-second';
        let badgeText = index === 0 ? '⭐ НАЙКРАЩИЙ РЕЖИМ' : '👍 ДРУГИЙ ВАРІАНТ';

        htmlContent += `
            <div class="${cardClass}">
                <div class="card-title">
                    <span>${v.color} форсунки</span>
                    <span class="${badgeClass}">${badgeText}</span>
                </div>
                <div class="data-row">
                    <span class="data-label">Робочий тиск:</span>
                    <span class="data-value">${v.pressure.toFixed(1)} бар</span>
                </div>
                <div class="data-row">
                    <span class="data-label">Витрата розпилювача:</span>
                    <span class="data-value">${v.flow.toFixed(3)} л/хв</span>
                </div>
                <div class="data-row" style="margin-top: 10px; border-top: 1px dashed #ccc; padding-top: 8px;">
                    <span class="data-label" style="color: #2c3e50; font-weight: bold;">ШВИДКІСТЬ ТРАКТОРА:</span>
                    <span class="data-value speed-highlight">${v.speed.toFixed(2)} км/год</span>
                </div>
            </div>
        `;
    });

    outputDiv.innerHTML = htmlContent;
}
